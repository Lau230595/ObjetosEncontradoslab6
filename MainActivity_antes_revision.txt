// =====================================================================
// 1. RUTAS: cada pantalla del flujo tiene un nombre único
// =====================================================================
sealed class Pantalla(val ruta: String) {
    object Inicio : Pantalla("inicio")
    object Reporte : Pantalla("reporte")
    object Revisar : Pantalla("revisar")
    object Confirmacion : Pantalla("confirmacion")
    object MisReportes : Pantalla("mis_reportes")
}

data class Reporte(val codigo: String, val tipo: String, val ubicacion: String, val estado: String)

// =====================================================================
// 2. NAVEGACIÓN: NavController (historial) + NavHost (ruta → pantalla)
// =====================================================================
@Composable
fun AppUnsaEpis() {
    val navController = rememberNavController()

    // Datos del formulario compartidos entre "reporte" y "revisar"
    var tipoObjeto by rememberSaveable { mutableStateOf("") }
    var ubicacion by rememberSaveable { mutableStateOf("") }
    var nota by rememberSaveable { mutableStateOf("") }

    // Reportes registrados (para "Mis reportes")
    val reportes = remember {
        mutableStateListOf(
            Reporte("#OE-20260915-102", "Documentos", "Biblioteca, 2.º piso", "Entregado")
        )
    }

    NavHost(navController = navController, startDestination = Pantalla.Inicio.ruta) {

        composable(Pantalla.Inicio.ruta) {
            PantallaInicioEPIS(
                onReportar = { navController.navigate(Pantalla.Reporte.ruta) },
                onMisReportes = { navController.navigate(Pantalla.MisReportes.ruta) }
            )
        }

        composable(Pantalla.Reporte.ruta) {
            PantallaReporteEPIS(
                tipoObjeto = tipoObjeto, ubicacion = ubicacion, nota = nota,
                onTipo = { tipoObjeto = it },
                onUbicacion = { ubicacion = it },
                onNota = { nota = it },
                onVolver = { navController.popBackStack() },
                onContinuar = { navController.navigate(Pantalla.Revisar.ruta) }
            )
        }

        composable(Pantalla.Revisar.ruta) {
            PantallaRevisarEPIS(
                tipoObjeto = tipoObjeto, ubicacion = ubicacion, nota = nota,
                onEditar = { navController.popBackStack() },
                onConfirmar = {
                    reportes.add(0, Reporte("#OE-20260929-204", tipoObjeto, ubicacion, "Pendiente de entrega"))
                    // popUpTo("inicio"): el formulario y la revisión salen del historial,
                    // así el usuario no puede regresar y enviar el mismo reporte dos veces.
                    navController.navigate(Pantalla.Confirmacion.ruta) {
                        popUpTo(Pantalla.Inicio.ruta)
                    }
                }
            )
        }

        composable(Pantalla.Confirmacion.ruta) {
            PantallaConfirmacionEPIS(
                codigo = "#OE-20260929-204",
                onMisReportes = {
                    tipoObjeto = ""; ubicacion = ""; nota = ""
                    navController.navigate(Pantalla.MisReportes.ruta) { popUpTo(Pantalla.Inicio.ruta) }
                },
                onInicio = {
                    tipoObjeto = ""; ubicacion = ""; nota = ""
                    navController.popBackStack(Pantalla.Inicio.ruta, inclusive = false)
                }
            )
        }

        composable(Pantalla.MisReportes.ruta) {
            PantallaMisReportesEPIS(
                reportes = reportes,
                onVolver = { navController.popBackStack() }
            )
        }
    }
}

/** Encabezado guinda con botón de regreso (‹), igual que en el Lab 04. */
@Composable
fun EncabezadoEPIS(titulo: String, onVolver: (() -> Unit)?) {
    Row(
        modifier = Modifier
            .fillMaxWidth()
            .background(EpisGuinda)
            .statusBarsPadding()
            .padding(horizontal = 8.dp, vertical = 12.dp),
        verticalAlignment = Alignment.CenterVertically
    ) {
        if (onVolver != null) {
            Text(
                "‹", color = Color.White, fontSize = 32.sp,
                modifier = Modifier.clickable { onVolver() }.padding(horizontal = 12.dp)
            )
        } else {
            Spacer(Modifier.size(16.dp))
        }
        Column {
            Text(titulo, color = Color.White, fontSize = 20.sp, fontWeight = FontWeight.Bold)
            Text("UNSA EPIS", color = Color.White.copy(alpha = 0.8f), fontSize = 12.sp)
        }
    }
}

/** Estructura de las pantallas internas: encabezado + contenido con scroll. */
@Composable
fun PantallaBaseEPIS(
    titulo: String,
    onVolver: (() -> Unit)?,
    contenido: @Composable ColumnScope.() -> Unit
) {
    Column(Modifier.fillMaxSize().background(EpisFondo)) {
        EncabezadoEPIS(titulo, onVolver)
        Column(
            modifier = Modifier
                .fillMaxSize()
                .verticalScroll(rememberScrollState())
                .navigationBarsPadding()
                .padding(20.dp),
            verticalArrangement = Arrangement.spacedBy(14.dp),
            content = contenido
        )
    }
}

@Composable
fun MensajeEPIS(texto: String, color: Color = EpisGuinda) {
    Card(
        colors = CardDefaults.cardColors(containerColor = EpisLila),
        shape = RoundedCornerShape(13.dp),
        modifier = Modifier.fillMaxWidth()
    ) {
        Text(texto, color = color, fontSize = 14.sp, modifier = Modifier.padding(14.dp))
    }
}

@Composable
fun ServicioEPIS(
    icono: String, titulo: String, descripcion: String,
    colorTarjeta: Color, modifier: Modifier = Modifier, onClick: () -> Unit
) {
    Card(
        modifier = modifier.clickable { onClick() },
        shape = RoundedCornerShape(20.dp),
        colors = CardDefaults.cardColors(containerColor = colorTarjeta)
    ) {
        Column(Modifier.padding(15.dp)) {
            Text(icono, fontSize = 27.sp)
            Spacer(Modifier.height(14.dp))
            Text(titulo, color = EpisTexto, fontWeight = FontWeight.Bold, fontSize = 14.sp)
            Spacer(Modifier.height(5.dp))
            Text(descripcion, color = EpisTextoSec, fontSize = 12.sp)
        }
    }
}

/** Pantalla 2 – Formulario "Reportar hallazgo" */
@Composable
fun PantallaReporteEPIS(
    tipoObjeto: String, ubicacion: String, nota: String,
    onTipo: (String) -> Unit, onUbicacion: (String) -> Unit, onNota: (String) -> Unit,
    onVolver: () -> Unit, onContinuar: () -> Unit
) {
    var mensaje by remember { mutableStateOf("") }

    PantallaBaseEPIS(titulo = "Reportar hallazgo", onVolver = onVolver) {
        Text("Ayúdanos a encontrar a su dueño 💜", color = EpisGuinda, fontSize = 18.sp, fontWeight = FontWeight.Bold)
        Text("Paso 1 de 2 · Completa los datos del objeto que encontraste.", color = EpisTextoSec, fontSize = 13.sp)

        // Área referencial de la fotografía
        Box(
            Modifier.fillMaxWidth().height(110.dp).background(EpisLila, RoundedCornerShape(20.dp)),
            contentAlignment = Alignment.Center
        ) {
            Column(horizontalAlignment = Alignment.CenterHorizontally) {
                Text("📷", fontSize = 28.sp)
                Text("Fotografía del objeto", color = EpisMorado, fontWeight = FontWeight.Bold)
                Text("Espacio referencial para la imagen", color = EpisTextoSec, fontSize = 11.sp)
            }
        }

        // 1. Tipo de objeto
        Text("1. Tipo de objeto *", color = EpisTexto, fontWeight = FontWeight.Bold)
        listOf("Documentos", "Dispositivo electrónico", "Ropa o accesorios", "Otro objeto").forEach { tipo ->
            Card(
                modifier = Modifier.fillMaxWidth(),
                shape = RoundedCornerShape(13.dp),
                colors = CardDefaults.cardColors(containerColor = if (tipoObjeto == tipo) EpisLila else Color.White),
                border = BorderStroke(1.dp, if (tipoObjeto == tipo) EpisMorado else EpisBorde)
            ) {
                Row(
                    Modifier.fillMaxWidth().clickable { onTipo(tipo); mensaje = "" }
                        .padding(horizontal = 10.dp, vertical = 3.dp),
                    verticalAlignment = Alignment.CenterVertically
                ) {
                    RadioButton(
                        selected = tipoObjeto == tipo,
                        onClick = { onTipo(tipo); mensaje = "" },
                        colors = RadioButtonDefaults.colors(selectedColor = EpisGuinda)
                    )
                    Text(tipo, color = EpisTexto, fontSize = 14.sp)
                }
            }
        }

        // 2. Ubicación
        Text("2. ¿Dónde lo encontraste? *", color = EpisTexto, fontWeight = FontWeight.Bold)
        OutlinedTextField(
            value = ubicacion,
            onValueChange = { onUbicacion(it); mensaje = "" },
            placeholder = { Text("Ej. Aula 204, pabellón de Sistemas") },
            singleLine = true, shape = RoundedCornerShape(13.dp),
            modifier = Modifier.fillMaxWidth()
        )

        // 3. Nota
        Text("3. Nota adicional (opcional)", color = EpisTexto, fontWeight = FontWeight.Bold)
        OutlinedTextField(
            value = nota, onValueChange = onNota,
            placeholder = { Text("Describe algún detalle del objeto...") },
            shape = RoundedCornerShape(13.dp),
            modifier = Modifier.fillMaxWidth().heightIn(min = 90.dp)
        )

        Button(
            onClick = {
                when {
                    tipoObjeto.isBlank() -> mensaje = "Selecciona el tipo de objeto."
                    ubicacion.isBlank() -> mensaje = "Completa la ubicación del hallazgo."
                    else -> { mensaje = ""; onContinuar() }
                }
            },
            modifier = Modifier.fillMaxWidth().height(48.dp),
            shape = RoundedCornerShape(13.dp)
        ) { Text("Continuar a revisión", fontWeight = FontWeight.Bold) }

        OutlinedButton(
            onClick = { mensaje = "Borrador guardado. Puedes continuar luego." },
            modifier = Modifier.fillMaxWidth().height(48.dp),
            shape = RoundedCornerShape(13.dp)
        ) { Text("Guardar borrador", color = EpisGuinda) }

        if (mensaje.isNotEmpty()) MensajeEPIS(mensaje)
    }
}

/** Pantalla 3 – Revisar antes de enviar */

    tipoObjeto: String, ubicacion: String, nota: String,
    onEditar: () -> Unit, onConfirmar: () -> Unit
) {
    PantallaBaseEPIS(titulo = "Revisar reporte", onVolver = onEditar) {
        Text("Paso 2 de 2 · Verifica los datos antes de enviar.", color = EpisTextoSec, fontSize = 13.sp)
        Card(
            shape = RoundedCornerShape(20.dp),
            colors = CardDefaults.cardColors(containerColor = Color.White),

                FilaDato("Nota", nota.ifBlank { "—" })

        }

            onClick = onConfirmar,

            modifier = Modifier.fillMaxWidth().height(48.dp),
            shape = RoundedCornerShape(13.dp)

/** Pantalla 5 – Mis reportes (el Estudiante 3 la amplía con el patrón lista-detalle) */
@Composable
fun PantallaMisReportesEPIS(reportes: List<Reporte>, onVolver: () -> Unit) {
    PantallaBaseEPIS(titulo = "Mis reportes", onVolver = onVolver) {
        if (reportes.isEmpty()) {
            MensajeEPIS("Aún no tienes reportes registrados.", EpisTextoSec)
        }
        reportes.forEach { r ->
            Card(
                shape = RoundedCornerShape(16.dp),
                colors = CardDefaults.cardColors(containerColor = Color.White),
                border = BorderStroke(1.dp, EpisBorde),
                modifier = Modifier.fillMaxWidth()
            ) {
                Column(Modifier.padding(16.dp), verticalArrangement = Arrangement.spacedBy(4.dp)) {
                    Text("${r.tipo} · ${r.ubicacion}", color = EpisTexto, fontWeight = FontWeight.Bold)
                    Text(r.codigo, color = EpisTextoSec, fontSize = 13.sp)
                    Text(
                        if (r.estado == "Entregado") "✓ Entregado" else "⏱ ${r.estado}",
                        color = if (r.estado == "Entregado") EpisExito else EpisMorado,
                        fontSize = 13.sp, fontWeight = FontWeight.Bold
                    )
                }
            }
        }
        OutlinedButton(
            onClick = onVolver,
            modifier = Modifier.fillMaxWidth().height(48.dp),
            shape = RoundedCornerShape(13.dp)
        ) { Text("Volver", color = EpisGuinda) }
    }
}
