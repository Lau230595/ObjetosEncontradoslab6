@Composable

    Column(
        modifier = modifier
            .fillMaxSize()
            .padding(24.dp),
        verticalArrangement = Arrangement.spacedBy(16.dp)
    ) {
        Text("Campus App", style = MaterialTheme.typography.bodyMedium)
        Text("Objetos encontrados", style = MaterialTheme.typography.headlineMedium)

            style = MaterialTheme.typography.bodyLarge

        Card(modifier = Modifier.fillMaxWidth()) {
            Column(Modifier.padding(16.dp)) {
                Text("Cómo funciona", style = MaterialTheme.typography.titleMedium)

            }
        }

            Text("Reportar objeto encontrado")
        }
        OutlinedButton(onClick = { }, modifier = Modifier.fillMaxWidth()) {
            Text("Mis reportes")
        }
    }
}
